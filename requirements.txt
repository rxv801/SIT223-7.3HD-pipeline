# Taskmaster — Python CV worker dependencies
# Built/tested on Python 3.11 (MediaPipe has no wheels for 3.13/3.14 yet).
#
# Installed for you by ./setup.sh. To do it manually:
#   cd python
#   python3.11 -m venv .venv
#   source .venv/bin/activate
#   pip install -r requirements.txt

# --- Computer vision (needed now: phone + gaze detection) ---
# opencv-python is deliberately NOT listed. mediapipe already depends on
# opencv-contrib-python, which is a superset of it, and installing both writes
# two copies into the same cv2/ package -- ~46MB of redundancy, and two sets of
# files that must be uninstalled together or the install is left broken.
# Letting mediapipe pull opencv-contrib-python alone keeps one clean install.
# Pinned, not floored: mediapipe 1.0.x aborts natively (DrishtiMetalHelper
# "Service is unavailable") when the face-detector subgraph initialises Metal
# in a headless process, which is how CI and the deployed worker both run it.
# 0.10.35 is the version the desktop app ships and the last one verified
# working headless on macOS arm64.
mediapipe==0.10.35          # face mesh for gaze detection
onnxruntime>=1.17           # runs the YOLOX phone-detection model (local, MIT)
numpy>=1.26,<2.0            # array math; pinned <2 for MediaPipe compatibility

# --- WebSocket server (needed later: Electron <-> Python bridge) ---
fastapi>=0.110              # app + routing for the detection-event server
uvicorn[standard]>=0.27     # ASGI server that runs FastAPI
websockets>=12.0            # WebSocket transport to the Electron main process

# --- Monitoring ---
prometheus-fastapi-instrumentator>=7.0   # exposes /metrics for Prometheus
prometheus-client>=0.21                  # custom counters for detector outcomes
